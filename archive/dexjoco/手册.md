# FastWAM 命令手册

## 1. 数据构造

```bash
# 双臂（例：microwave_cook，action=44, state=46，3 视角）
python scripts/prepare_dexjoco_ego_dataset.py \
  --source-root data/dexjoco/dexjoco_lerobot_datasets \
  --output-root data/dexjoco_microwave_cook \
  --include-substr microwave_cook \
  --overwrite

# 单臂（例：water_plant，action=22, state=23，1 视角）
python scripts/prepare_dexjoco_ego_dataset.py \
  --source-root data/dexjoco/dexjoco_lerobot_datasets \
  --output-root data/dexjoco_water_plant \
  --include-substr water_plant \
  --single-arm \
  --overwrite
```

## 2. 计算 text_emb

```bash
# 单卡预计算 T5 embedding cache
python scripts/precompute_text_embeds.py task=dexjoco_microwave_cook_uncond_3cam_384_1e-4
python scripts/precompute_text_embeds.py task=dexjoco_water_plant_uncond_1cam_384_1e-4

# 多卡预计算（以双臂为例）
torchrun --standalone --nproc_per_node=8 \
  scripts/precompute_text_embeds.py task=dexjoco_microwave_cook_uncond_3cam_384_1e-4

# 评测前导出 .npz（eval 脚本也会自动触发）
python scripts/export_text_embed_cache_npz.py \
  --cache-dir data/text_embeds_cache/dexjoco_microwave_cook
```

## 3. 训练

```bash
# 训练前生成 ActionDiT backbone（首次必做）
python scripts/preprocess_action_dit_backbone.py \
  --model-config configs/model/fastwam.yaml \
  --output checkpoints/ActionDiT_linear_interp_Wan22_alphascale_1024hdim.pt \
  --device cuda \
  --dtype bfloat16

# 双臂
bash scripts/train_zero1.sh 8 task=dexjoco_microwave_cook_uncond_3cam_384_1e-4

# 单臂
bash scripts/train_zero1.sh 4 task=dexjoco_water_plant_uncond_1cam_384_1e-4
```

## 4. 测试

```bash
# 闭环评测：先启 policy server（fastwam 环境）

# 双臂
CUDA_VISIBLE_DEVICES=7 python scripts/run_fastwam_server.py \
  --run-dir runs/dexjoco_microwave_cook_uncond_3cam_384_1e-4/<run_id> \
  --checkpoint runs/dexjoco_microwave_cook_uncond_3cam_384_1e-4/<run_id>/step_XXXXX.pt \
  --device cuda:0 --host 0.0.0.0 --port 5560

# 单臂
CUDA_VISIBLE_DEVICES=7 python scripts/run_fastwam_server.py \
  --run-dir runs/dexjoco_water_plant_uncond_1cam_384_1e-4/<run_id> \
  --checkpoint runs/dexjoco_water_plant_uncond_1cam_384_1e-4/<run_id>/step_XXXXX.pt \
  --device cuda:0 --host 0.0.0.0 --port 5560

# 闭环评测：再跑 eval（dexjoco 环境）
bash scripts/run_eval_dexjoco_fastwam.sh
TASKS=water_plant EPISODES=50 bash scripts/run_eval_dexjoco_fastwam.sh
```
